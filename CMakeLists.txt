cmake_minimum_required(VERSION 3.26)
project(PocketVR VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
include(CTest)
option(POCKETVR_BUILD_WINDOWS "Build Windows driver and applications" ON)
add_library(pocketvr_options INTERFACE)
if(MSVC)
  target_compile_options(pocketvr_options INTERFACE /W4 /WX /permissive- /EHsc /utf-8)
  target_compile_definitions(pocketvr_options INTERFACE NOMINMAX WIN32_LEAN_AND_MEAN)
else()
  target_compile_options(pocketvr_options INTERFACE -Wall -Wextra -Wpedantic -Werror)
endif()
find_package(Python3 COMPONENTS Interpreter REQUIRED)
add_custom_target(license-audit
  COMMAND "${Python3_EXECUTABLE}" "${PROJECT_SOURCE_DIR}/scripts/license_audit.py"
  WORKING_DIRECTORY "${PROJECT_SOURCE_DIR}" VERBATIM)
if(BUILD_TESTING)
  add_test(NAME license-policy COMMAND "${Python3_EXECUTABLE}" "${PROJECT_SOURCE_DIR}/scripts/license_audit.py")
  add_test(NAME license-audit-tests COMMAND "${Python3_EXECUTABLE}" -m unittest discover -s "${PROJECT_SOURCE_DIR}/scripts/tests" -v)
  add_test(NAME settings-generated COMMAND "${Python3_EXECUTABLE}" "${PROJECT_SOURCE_DIR}/scripts/generate_settings.py" --check)
endif()
if(EXISTS "${PROJECT_SOURCE_DIR}/pc/common/CMakeLists.txt")
  add_subdirectory(pc/common)
endif()
if(WIN32 AND POCKETVR_BUILD_WINDOWS)
  foreach(component tools driver control overlay)
    if(EXISTS "${PROJECT_SOURCE_DIR}/pc/${component}/CMakeLists.txt")
      add_subdirectory("pc/${component}")
    endif()
  endforeach()
endif()

if(TARGET pocketvr_control AND TARGET pocketvr_overlay)
  add_dependencies(pocketvr_control pocketvr_overlay)
  add_subdirectory(pc/installer)
endif()
