cmake_minimum_required(VERSION 3.26)
project(PocketVRAndroid LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(NOT ANDROID_ABI STREQUAL "arm64-v8a")
  message(FATAL_ERROR "PocketVR currently supports arm64-v8a only")
endif()
get_filename_component(POCKETVR_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../../../.." ABSOLUTE)
set(POCKETVR_NATIVE_PREFIX "${VCPKG_INSTALLED_DIR}/arm64-pocketvr-android")
# The Android Gradle plugin owns the NDK toolchain. Some vcpkg package
# configs also need these variables when consumed without vcpkg's toolchain.
set(_VCPKG_INSTALLED_DIR "${VCPKG_INSTALLED_DIR}")
set(VCPKG_TARGET_TRIPLET "arm64-pocketvr-android")
list(PREPEND CMAKE_PREFIX_PATH "${POCKETVR_NATIVE_PREFIX}")
list(APPEND CMAKE_FIND_ROOT_PATH "${POCKETVR_NATIVE_PREFIX}")
option(POCKETVR_NATIVE_TESTS "Include the native test entry point in a development build" OFF)
set(BUILD_TESTING OFF)
add_library(pocketvr_options INTERFACE)
target_compile_options(pocketvr_options INTERFACE -Wall -Wextra -Wpedantic -Werror)
add_subdirectory("${POCKETVR_ROOT}/pc/common" "${CMAKE_CURRENT_BINARY_DIR}/common")
find_package(nlohmann_json CONFIG REQUIRED)
target_sources(pocketvr_common PRIVATE "${POCKETVR_ROOT}/pc/common/src/settings.cpp")
target_link_libraries(pocketvr_common PUBLIC nlohmann_json::nlohmann_json)
add_library(pocketvr_native SHARED client.cpp discovery.cpp sensors.cpp decoder.cpp renderer.cpp)
target_link_libraries(pocketvr_native PRIVATE pocketvr_common android log mediandk EGL GLESv3 dl)
target_link_options(pocketvr_native PRIVATE -Wl,-z,max-page-size=16384 -Wl,--gc-sections)
if(POCKETVR_NATIVE_TESTS)
  target_compile_definitions(pocketvr_native PRIVATE POCKETVR_NATIVE_TESTS=1)
  find_package(GTest CONFIG REQUIRED)
  target_sources(pocketvr_native PRIVATE native_tests.cpp renderer_tests.cpp
    "${POCKETVR_ROOT}/pc/common/tests/common_tests.cpp"
    "${POCKETVR_ROOT}/pc/common/tests/pairing_tests.cpp"
    "${POCKETVR_ROOT}/pc/common/tests/session_tests.cpp")
  target_sources(pocketvr_native PRIVATE "${POCKETVR_ROOT}/pc/common/tests/input_tests.cpp")
  target_sources(pocketvr_native PRIVATE "${POCKETVR_ROOT}/pc/common/tests/stream_format_tests.cpp")
  target_link_libraries(pocketvr_native PRIVATE GTest::gtest)
endif()
